import { Link, useLocation } from "react-router-dom";
import { User } from "lucide-react";
import { Logo } from "@/components/logo";

export const Navbar = () => {
  const location = useLocation();

  const isActive = (path: string) => location.pathname === path;

  const navItems = [
    { name: "Education", path: "/lms-main" },
    { name: "Stock Research", path: "/stock-research" },
    { name: "Event", path: "/event" },
    { name: "Referral", path: "/referral" },
  ];

  return (
    <nav className="bg-[#081637] sticky top-0 z-50 border-b border-[#0d1c3a]">
      <div className="container mx-auto px-10 py-4 flex items-center justify-between">
        {/* Logo */}
        <Link to="/" className="flex items-center gap-2">
          <Logo size={28} />
          <span className="text-lg font-bold text-white tracking-wide">
            STOCKWISE
          </span>
        </Link>

        {/* Navigation Links */}
        <div className="flex items-center gap-10">
          {navItems.map((item) => (
            <Link
              key={item.path}
              to={item.path}
              className={`relative text-sm font-medium transition-colors ${
                isActive(item.path)
                  ? "text-white after:absolute after:bottom-[-4px] after:left-0 after:w-full after:h-[2px] after:bg-[#0070F3]"
                  : "text-gray-400 hover:text-white"
              }`}
            >
              {item.name}
            </Link>
          ))}
        </div>

        {/* User */}
        <button className="flex items-center gap-2 text-[#0070F3] hover:text-blue-400 transition-colors">
          <User className="w-4 h-4" />
          <span className="text-sm font-medium">username</span>
        </button>
      </div>
    </nav>
  );
};
